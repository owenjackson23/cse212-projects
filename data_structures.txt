public class DataStructures

Dynamic Array (List)
{
    Implementation Example: 
        var numbers = new List<int>();

    purpose = An array that does not have a fixed size

    pros = [Expandable]
    cons = [
        More memory,
        slower memory allocation
    ]

    commonOperations = [
        lookup: value = myList[index]
        append: myList.Add(value)
        insert: myList.Insert(index, value)
        remove: myList.Remove(index)
        size: myList.Count
        capacity: myList.Capacity
        empty: myList.Count == 0
    ]
}

Stack - "Last In, First Out" (LIFO)
{
    Implementation Example:
        var Stack = new Stack<int>(); 

    purpose = The stack is used to reverse data
        + or remember previous data (including previous results).
        + It follows a "Last In, First Out" (LIFO) rule
    
    pros = [
        remembering previous data,
        performing an operation backwards
    ]
    cons = [
        some data may not be used for awhile
    ]

    commonOperations = [
        push: myStack.Push(value)
            putting something onto the stack
        pop: var value = myStack.Pop()
            taking something off of a stack
        size: var length = myStack.Count
        empty: if (myStack.Count == 0)
    ]

    otherDefinitions = [
        back: the location in a stack where a push and pop occurs.
            + The last item put into the stack is found in the back.
        front: the location in a stack where the first item added to the stack can be found.
            + Traditionally, this is the index 0 of a dynamic array.
    ]
}

Queue - "First In, First Out" (FIFO)
{
    Implementation Example:
    var q = new Queue<char>();

    purpose = processing requests in a fair and orderly way

    pros = [
        Web server queues,
        Reader/Writer queues
    ]
    cons = [
        Harder to work backwards
    ]

    commonOperations = [
        enqueue: myQueue.Enqueue(value)
            add an item to the back of the queue
        dequeue: var value = myQueue.Dequeue()
            removes an item from the front of the queue
        size: myQueue.Count
        empty: if (myQueue.Count == 0)
    ]

    otherDefinitions = [
        back: the location in the queue where an enqueue occurs.
            + The last item put in the queue is found in the back.
        front: the location in the queue where a dequeue occurs.
            + The first item put in the queue is found in the front.
            + If you use a dynamic array (List) this is index 0.
    ]
}

Set
{
    Implementation Examples:
    var set1 = new HashSet(){1, 2, 3, 4, 5};
    var found = new HashSet<char>();

    purpose = storing unique values
        + quickly searching for unique values

    pros = [
        order does not matter,
        fast performance for adding + removing + finding (testing for membership),
        quick access to unique results previously calculated,
        performing mathematical set operations such as an intersection an union
    ]
    cons = [
        does not allow duplicates
    ]

    commonOperations = [
        positiveIntHashCode: 3.GetHashCode()
            returns a hash code of 3
        negativeIntHashCode: -3.GetHashCode()
            returns a hash code of -3
        stringHashCode: "cat".GetHashCode()
            returns a hash code of -1599535192
        floatDoubleHashCode: 3.14.GetHashCode()
            returns a hash code of 300063655
        boolHashCode: true.GetHashCode()
            returns a hash code of 1
        listObjectHashCode: new List<string>() .GetHashCode()
            returns a hash code of 27252167
        
        add: mySet.Add(value)
        remove: mySet.Remove(value)
        member: mySet.Contains(value)
        size: mySet.Count
    ]

    otherDefinitions = [
        hashing: the process of mapping an item to an index location using a hashing function
        hashingFunction: a function that converts the value of an item to a numerical index value.
            + Includes a modulo operation
        sparseArray: an array that is only partially filled.
            + To avoid conflicts in a set, a sparse array must have sufficient empty space to allow for new additions
        openAddressing: A method of removing conflicts in a set in which a new empty location is found elsewhere in the sparse array.
        chaining: a method of removing conflicts in a set.
            + All items that hash to the same index are chained together into a single data structure stored in that target index
    ]
}

Map (AKA HashTable, AKA HashMap, AKA Dictionary)
{
    Implementation Examples:
    var dictionary = new Dictionary<string, double>();
    var dictionary = new Dictionary<string, double>() { { "key1", 2.1 }, { "key2", 2.3 } };

    purpose = similar to a set, but uses key-value pairs

    pros = [
        quick lookup for finding values
    ]
    cons = []

    commonOperations = [
        put: myMap.Add(key, value)
             myMap[key] = value
            adds (or replaces) a row
        get: myMap[key]
            gets the value for the specified key
        remove: myMap.Remove(key)
            removes the row from the map containing the specified key
        member: myMap.ContainsKey(key)
            determines if "key" is in the map
        size: myMap.Count
            returns the number of items in the map
    ]

    otherDefinitions = [
        key: keys are used to lookup values from the map.
            + The keys in the map form a set.
            + Each key is unique.
        value: the value associated with each key within a map.
            + Often referred to as key-value pairs.
        JSON: JavaScript Object Notation.
            + A format used frequently to share data between software.
            + JSON data uses maps and lists.
            + The syntax of JSON is the same as Python.
    ]
}
